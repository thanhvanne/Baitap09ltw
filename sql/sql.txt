CREATE TABLE roles
(
    id BIGINT IDENTITY(1,1) NOT NULL,
    name VARCHAR(30) NOT NULL,

    CONSTRAINT PK_roles
        PRIMARY KEY (id),

    CONSTRAINT UQ_roles_name
        UNIQUE (name)
);
GO


CREATE TABLE users
(
    id BIGINT IDENTITY(1,1) NOT NULL,

    username VARCHAR(50) NOT NULL,

    email VARCHAR(150) NOT NULL,

    password VARCHAR(150) NOT NULL,

    full_name NVARCHAR(150) NOT NULL,

    images VARCHAR(500) NULL,

    enabled BIT NOT NULL
        CONSTRAINT DF_users_enabled
        DEFAULT 1,

    created_at DATETIME2 NOT NULL
        CONSTRAINT DF_users_created_at
        DEFAULT SYSDATETIME(),

    role_id BIGINT NOT NULL,

    CONSTRAINT PK_users
        PRIMARY KEY (id),

    CONSTRAINT UQ_users_username
        UNIQUE (username),

    CONSTRAINT UQ_users_email
        UNIQUE (email),

    CONSTRAINT FK_users_roles
        FOREIGN KEY (role_id)
        REFERENCES roles(id)
);
GO


CREATE TABLE products
(
    id BIGINT IDENTITY(1,1) NOT NULL,

    name NVARCHAR(200) NOT NULL,

    description NVARCHAR(1000) NULL,

    price DECIMAL(18,2) NOT NULL,

    image VARCHAR(500) NULL,

    image_public_id VARCHAR(500) NULL,

    created_at DATETIME2 NOT NULL
        CONSTRAINT DF_products_created_at
        DEFAULT SYSDATETIME(),

    user_id BIGINT NOT NULL,

    CONSTRAINT PK_products
        PRIMARY KEY (id),

    CONSTRAINT CK_products_price
        CHECK (price >= 0),

    CONSTRAINT FK_products_users
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
);
GO


CREATE TABLE otp_tokens
(
    id BIGINT IDENTITY(1,1) NOT NULL,

    otp_hash VARCHAR(100) NOT NULL,

    type VARCHAR(30) NOT NULL,

    expires_at DATETIME2 NOT NULL,

    attempts INT NOT NULL
        CONSTRAINT DF_otp_tokens_attempts
        DEFAULT 0,

    used BIT NOT NULL
        CONSTRAINT DF_otp_tokens_used
        DEFAULT 0,

    created_at DATETIME2 NOT NULL
        CONSTRAINT DF_otp_tokens_created_at
        DEFAULT SYSDATETIME(),

    user_id BIGINT NOT NULL,

    CONSTRAINT PK_otp_tokens
        PRIMARY KEY (id),

    CONSTRAINT CK_otp_tokens_attempts
        CHECK (attempts >= 0),

    CONSTRAINT FK_otp_tokens_users
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE
);
GO


CREATE INDEX IX_users_full_name
ON users(full_name);
GO

CREATE INDEX IX_users_role_id
ON users(role_id);
GO

CREATE INDEX IX_products_name
ON products(name);
GO

CREATE INDEX IX_products_user_id
ON products(user_id);
GO

CREATE INDEX IX_products_created_at
ON products(created_at);
GO

CREATE INDEX IX_otp_user_type
ON otp_tokens(user_id, type);
GO

CREATE INDEX IX_otp_created_at
ON otp_tokens(created_at DESC);
GO

SELECT
    r.id,
    r.name
FROM roles r
ORDER BY r.id;
GO


SELECT
    u.id,
    u.username,
    u.email,
    u.full_name,
    u.enabled,
    u.created_at,
    r.name AS role_name,
    COUNT(p.id) AS product_count
FROM users u
INNER JOIN roles r
    ON r.id = u.role_id
LEFT JOIN products p
    ON p.user_id = u.id
GROUP BY
    u.id,
    u.username,
    u.email,
    u.full_name,
    u.enabled,
    u.created_at,
    r.name
ORDER BY u.id;
GO


SELECT
    p.id,
    p.name,
    p.description,
    p.price,
    p.image,
    p.image_public_id,
    p.created_at,
    u.username AS owner_username,
    u.full_name AS owner_name
FROM products p
INNER JOIN users u
    ON u.id = p.user_id
ORDER BY p.id DESC;
GO

    COUNT(*) AS total_users

    COUNT(*) AS total_products

    u.username,
    u.full_name,
    COUNT(p.id) AS product_count
FROM users u
LEFT JOIN products p
    ON p.user_id = u.id
GROUP BY
    u.username,
    u.full_name
ORDER BY
    product_count DESC,
    u.username;
